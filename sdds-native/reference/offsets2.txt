void CHUDXMLRadioMsg::Reconstruct()
{
	if(IsVisible())
	{
		char szTemp[MAX_PATH];
		std::wstring wstrTemp;
 
		if(m_nPage == eRadioMsg_Emotion)
		{
			wstrTemp = g_pInterfaceResMgr->GetTextControlStringValueW("$Emotion_listClass_0").c_str();
			m_pTitle->SetText( wstrTemp );
 
			for(int i=0; i<RADIO_MESSAGE_MAX; i++)
			{
				if(m_pTextArray[i])
				{
					sprintf_s(szTemp, "Emotion_list_%d", i+1);
					wstrTemp = g_pInterfaceResMgr->GetTextControlStringValueW(szTemp).c_str();
 
					std::wstring wstrComp;
					wstring_from_string(szTemp, wstrComp);
 
					if(wcscmp(wstrTemp.c_str(), wstrComp.c_str()))
					{
						m_pTextArray[i]->SetText(wstrTemp);
						m_pTextArray[i]->SetVisible(LTTRUE);
					}
					else
						m_pTextArray[i]->SetVisible(LTFALSE);
				}
			}
 
			return;
		}
 
		strcpy(szTemp,g_pRadioMsgMgr->GetMsgClassName(m_nPage));
		wstring_from_string(szTemp, wstrTemp);
		m_pTitle->SetText( wstrTemp );
 
		for(int i=0; i<RADIO_MESSAGE_MAX; i++)
		{
			if(m_pTextArray[i])
			{
				if(m_nViewIdx + i < g_pRadioMsgMgr->GetMsgNumOfClass(m_nPage))
				{
					strcpy(szTemp,g_pRadioMsgMgr->GetMsgElementText(m_nPage,m_nViewIdx + i));
					wstring_from_string(szTemp, wstrTemp);
					m_pTextArray[i]->SetText(wstrTemp);
					m_pTextArray[i]->SetVisible(LTTRUE);
				}
				else
					m_pTextArray[i]->SetVisible(LTFALSE);
			}
		}
	}
}
 
void CHUDXMLRadioMsg::ShowRadioMsg(eKindRadioMsg eKind)
{
	if(g_pCBAPlayerMgr->GetPlayerState() != PS_ALIVE)
	{
		SetVisible(LTFALSE);
		return;
	}
 
	int nTick=GetTickCount();
	if(nTick - m_nLastCommandTick < s_nSoundPlayDelayTime)
		return ;
 
	if(!m_bBuilt)
		return;
 
	if(eKind >= eRadioMsg_MaxNum)
		return;
 
	if(IsVisible() && m_nPage==eKind)
	{
		SetVisible(LTFALSE);
	}
	else
	{
		m_nPage=eKind;
		m_nViewIdx=0;
		SetVisible(LTTRUE);
	}
 
	Reconstruct();
}
