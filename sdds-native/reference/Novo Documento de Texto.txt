*/
using show_message_box_fn = void (__thiscall*)( c_xml_popup_mgr*, const wchar_t* p_string, const c_mb_create& p_create, std::uint8_t font_size, std::uint32_t default_return );
...
void show_message_box( const std::wstring_view& string, const c_mb_create& mb )
{

}

void __cdecl load_string_w::hook(
	std::int32_t message_code,
	wchar_t* output,
	std::int32_t max_output_len
)
{
	/*
	   this is our custom caption id
	*/
	if ( message_code <= -1 )
	{
		/*
		   think a bit how you can get a string using a negative index and you will be fine here :)
		*/
		constexpr std::wstring_view caption{ L"Our caption" };
 
		/*
		   caption data and size
		*/
		const auto caption_data{ caption.data( ) };
		const auto caption_size{ caption.size( ) };
 
		/*
		   the output is an array of 256 wchar_t
		*/
		if ( caption_size < max_output_len )
		{ 
			wcsncpy_s( 
				output,
				max_output_len, 
				caption_data,
				caption_size
			);
 
			return;
		}
 
		/*
		   if we can't handle the custom value,
		   let's just reset it to IDS_CUSTOM
		*/
		message_code = 0;
	}
 
	/*
	   it's not a custom caption we're looking for, let's continue
	*/
	original( message_code, output, max_output_len );
}
